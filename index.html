      }

    function renderChoices(currentWord, answerProp) {

      const optionsGrid = document.getElementById('quiz-options-grid');

      optionsGrid.innerHTML = '';


      let wrongCandidates = MASTER_VOCAB.filter(item => item.k !== currentWord.k);

      let distractors = shuffle(wrongCandidates).slice(0, 3);

      let choices = shuffle([currentWord, ...distractors]);


      choices.forEach(choice => {

        const btn = document.createElement('button');

        btn.className = "w-full text-left p-4 bg-white border-2 border-slate-200/90 rounded-2xl font-medium text-slate-800 hover:border-brand-500 hover:bg-brand-50/40 active:scale-[0.98] transition shadow-sm leading-relaxed";

        btn.textContent = choice[answerProp];

        btn.onclick = () => handleChoiceAnswer(btn, choice, currentWord);

        optionsGrid.appendChild(btn);

      });

    }


    function handleChoiceAnswer(btn, chosen, correct) {

        feedbackMsg.innerHTML = '<span class="text-emerald-600"><i class="fa-solid fa-circle-check mr-1.5"></i>正解です！</span>';

      } else {

        btn.className = "w-full text-left p-4 bg-rose-50 border-2 border-rose-500 rounded-2xl font-bold text-rose-800 shadow-sm transition";

        const answerText = quizMode === 'ko-ja' ? correct.j : correct.k;

        feedbackMsg.innerHTML = `<span class="text-rose-600"><i class="fa-solid fa-circle-xmark mr-1.5"></i>不正解 (正解: ${answerText})</span>`;

        wrongAnswers.push(correct);


        allBtns.forEach(b => {

          const targetText = quizMode === 'ko-ja' ? correct.j : correct.k;

          if (b.textContent === targetText) {

            b.className = "w-full text-left p-4 bg-emerald-50 border-2 border-emerald-500 rounded-2xl font-bold text-emerald-800 transition";

          }

        });

      }


      playAudio(correct.k);

      document.getElementById('quiz-next-btn').classList.remove('hidden');

    }


    function submitTypingAnswer() {

      const input = document.getElementById('quiz-typing-input');

      const val = input.value.trim();

      if (!val) return;


      input.disabled = true;

      const cur = currentQuestions[currentIndex];

      const normalize = s => s.replace(/[\?\!\-\s\(\)\/]/g, '');

      const isCorrect = normalize(val) === normalize(cur.k);

      const feedbackMsg = document.getElementById('quiz-feedback-msg');


      if (isCorrect) {

        score++;

        feedbackMsg.innerHTML = '<span class="text-emerald-600"><i class="fa-solid fa-circle-check mr-1.5"></i>正解です！</span>';

      } else {

        feedbackMsg.innerHTML = `<span class="text-rose-600"><i class="fa-solid fa-circle-xmark mr-1.5"></i>不正解 (正解: ${cur.k})</span>`;

        wrongAnswers.push(cur);

      }


      playAudio(cur.k);

      document.getElementById('quiz-next-btn').classList.remove('hidden');

    }


    document.getElementById('quiz-typing-input').addEventListener('keydown', (e) => {

      if (e.key === 'Enter') submitTypingAnswer();

    });


    function nextQuestion() {

      currentIndex++;

      if (currentIndex < currentQuestions.length) {

        renderQuestion();

      } else {

        showResults();

      }

    }


    function showResults() {

      let msg = "もう少し練習して満点を目指しましょう！";

      let iconHtml = '<i class="fa-solid fa-award"></i>';

      let iconColor = 'bg-brand-100 text-brand-600';

        msg = "満点です！パーフェクト！完璧に定着しています！🎉";

        iconHtml = '<i class="fa-solid fa-crown text-amber-500"></i>';

        iconColor = 'bg-amber-100';

        msg = "合格レベルです！素晴らしい語彙力です！👍";

        iconHtml = '<i class="fa-solid fa-star text-brand-500"></i>';

        iconColor = 'bg-pink-100';

      } else if (rate >= 0.5) {

        msg = "あと少し！間違えた単語をしっかり復習しましょう！";

      }


      document.getElementById('result-message').textContent = msg;

      const trophyEl = document.getElementById('result-trophy-icon');

      trophyEl.className = `w-16 h-16 rounded-2xl flex items-center justify-center text-3xl mx-auto mb-3 shadow-inner ${iconColor}`;

      trophyEl.innerHTML = iconHtml;


      const mistakesContainer = document.getElementById('result-mistakes-container');

      if (wrongAnswers.length > 0) {

        mistakesContainer.innerHTML = wrongAnswers.map(item => `

          <div class="py-2.5 flex items-center justify-between">

          </div>

      }

    }
